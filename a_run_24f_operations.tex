\section*{Autoencoder Operations for a\_run\_24f (Tied Weights)}

\subsection*{Architecture Parameters}
\begin{itemize}
    \item Ambient dimension: 12
    \item Embedded dimension: 2
    \item Hidden dimensions: [24, 20, 12]
    \item Activation: smooth\_leaky\_relu with $a = 0.5$, $\epsilon = 1.0$
    \item Tied weights: true (decoder uses transpose of encoder weights)
    \item Bias: true
    \item RMS norm: false
\end{itemize}

\subsection*{Weight Matrices and Biases}
\begin{align*}
    W_1 &\[0.5em]
    W_2 &\[0.5em]
    W_3 &\[0.5em]
    W_4 &
\end{align*}

\subsection*{Encoder (Forward Pass)}

Given input $x \in \mathbb{R}^{12}$:

\begin{align*}
    h_1 &= \text{smooth\_leaky\_relu}(x W_1 + b_1) \\
    h_2 &= \text{smooth\_leaky\_relu}(h_1 W_2 + b_2) \\
    h_3 &= \text{smooth\_leaky\_relu}(h_2 W_3 + b_3) \\
    z &= h_3 W_4 + b_4
\end{align*}

where $z \in \mathbb{R}^2$ is the latent code.

\subsection*{Activation Functions}

\begin{align*}
    \text{smooth\_leaky\_relu}(x) &= \frac{1}{2} \left((1 + a)x + (1 - a)\sqrt{x^2 + \epsilon^2}\right) \\
    \text{smooth\_leaky\_relu\_inv}(y) &= \frac{(1 + a)y - (1 - a)\sqrt{y^2 + a \epsilon^2}}{2a}
\end{align*}

with $a = 0.5$ and $\epsilon = 1.0$.

\subsection*{Decoder (Inverse Pass with Tied Weights)}

Given latent code $z \in \mathbb{R}^2$:

\begin{align*}
    h_3' &= (z - b_4) W_4^T \\
    h_2' &= \text{smooth\_leaky\_relu\_inv}(h_3') \\
    h_2' &= (h_2' - b_3) W_3^T \\
    h_1' &= \text{smooth\_leaky\_relu\_inv}(h_2') \\
    h_1' &= (h_1' - b_2) W_2^T \\
    x' &= \text{smooth\_leaky\_relu\_inv}(h_1') \\
    x' &= (x' - b_1) W_1^T
\end{align*}

where $x' \in \mathbb{R}^{12}$ is the reconstruction.

\subsection*{Full Autoencoder}

\begin{equation*}
    x' = \text{decode}(\text{encode}(x))
\end{equation*}

\subsection*{Matrix Dimensions}

\begin{tabular}{ll}
    $W_1$: & $24 \times 12$ \\
    $b_1$: & $24$ \\
    $W_2$: & $20 \times 24$ \\
    $b_2$: & $20$ \\
    $W_3$: & $12 \times 20$ \\
    $b_3$: & $12$ \\
    $W_4$: & $2 \times 12$ \\
    $b_4$: & $2$ \\
\end{tabular}
